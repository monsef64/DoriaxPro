1 — Hello scene

Starter

Create a camera, sun and a floor — no editor scene needed.

lua/main.lua
local scene = Scene.new()

local cam = Camera.new(scene)
cam.position = Vector3.new(0, 3, 6)
cam:setPerspective(60, 0.1, 100)
cam:activate()

local light = Light.new(scene)
light:setDirection(-0.6, -1, -0.3)
light.color = Vector3.new(1, 0.98, 0.92)
light.intensity = 1.1

local ground = Mesh.new(scene)
ground:setMesh("plane")
ground.scale = Vector3.new(10, 1, 10)
ground.position = Vector3.new(0, 0, 0)

Engine:setScene(scene)

2 — Sprite with billboard

Rendering
lua
local scene = Scene.new()
local cam = Camera.new(scene); cam:activate()

local hero = Sprite.new(scene)
hero:setTexture("hero_idle.png")
hero.position = Vector3.new(0, 1, 0)
hero:setBillboard(true, true, false) -- always face camera
hero:setAnchorPreset(AnchorPreset.CENTER)

-- sprite-sheet animation
local anim = SpriteAnimation.new(scene, hero)
anim:setAnimation({
  {frame = "hero_0.png", duration = 0.08},
  {frame = "hero_1.png", duration = 0.08},
  {frame = "hero_2.png", duration = 0.08}
})
anim.loop = true

Engine:setScene(scene)

3 — Click to place + orbit camera

Input
lua — input & picking
local scene = Scene.new()
local cam = Camera.new(scene)
cam.position = Vector3.new(6, 4, 6)
cam:setTarget(0,0,0)
cam:activate()
Engine:setScene(scene)

Engine.onUpdate = function()
  -- orbit with mouse drag
  if Input:isMouseButtonDown(0) then
    cam:rotatePosition( Input.mouseDelta.x * 0.4, Input.mouseDelta.y * 0.4 )
  end
  cam:zoom( Input.mouseWheel * 0.8 ) -- wheel helper
end

Engine.onMouseDown = function(btn, x, y)
  if btn == 0 then
    local ray = cam:screenToRay(Vector2.new(x,y))
    local hit = scene:raycast(ray, RayFilter.ALL)
    if hit then Log:info("hit %s", hit.object:getName()) end
  end
end

4 — Physics drop (Box2D)

Physics
lua
local scene = Scene.new()
local cam = Camera.new(scene); cam:activate(); Engine:setScene(scene)

-- ground (static)
local ground = Body2D.new(scene)
ground.bodyType = BodyType.STATIC
ground.position = Vector3.new(0, -2, 0)
ground:setShape(Shape2DType.RECT, 12, 0.6)

-- stack of boxes (dynamic)
for i = 0, 4 do
  local b = Body2D.new(scene)
  b.bodyType = BodyType.DYNAMIC
  b.position = Vector3.new(0, i*1.2 + 1, 0)
  b:setShape(Shape2DType.RECT, 1, 1)
  b.linearVelocity = Vector2.new((math.random()-0.5)*2, 0)
end

-- listen for hits
ground.onBeginContact:add("log", function(a, b, m)
  Log:info("ground hit")
end)

5 — Tween anything with Ease

Actions
lua — ease + action
local scene = Scene.new()
local box = Mesh.new(scene); box:setMesh("cube")
Engine:setScene(scene)

local jump = PositionAction.new(scene, box)
jump:setAction(Ease.easeOutBounce(), Vector3.new(0,0,0), Vector3.new(0,3,0))
jump.duration = 0.6
jump.loop = false

Engine.onKeyDown = function(key)
  if key == Key.SPACE then jump:start() end
end

-- custom curve
local custom = Ease.new(EaseType.CUSTOM)
custom:setFunction(function(t) return 1 - (1-t)^3 end)

6 — Particles explosion

FX
lua
local scene = Scene.new()
Engine:setScene(scene)

local boom = Particles.new(scene)
boom.position = Vector3.new(0,1,0)
boom.rate = 0
boom:setLifeInitializer(0.4, 0.9)
boom:setVelocityInitializer(5, 9, Vector3.new(0,1,0), math.pi)
boom:setColorInitializer(Vector4.new(1,0.8,0.1,1), Vector4.new(1,0.2,0.0,1))
boom:addColorGradientStop(0.0, Vector4.new(1,1,0,1))
boom:addColorGradientStop(1.0, Vector4.new(1,0,0,0))
boom:setSizeInitializer(0.15, 0.35)

function explode(pos)
  boom.position = pos
  boom:addBurst(0, 40)
end

Engine.onMouseDown = function() explode(Vector3.new(0,1,0)) end

8 — Async scene flow with loading screen

Scene
lua — loading
Engine.asyncLoading = true
SceneManager.setLoadingScene("LoadingScreen") -- scene id shown while loading

function goToLevel2()
  SceneManager.preloadScene("Level2")
  SceneManager.holdLoading(true) -- keep cover up even when ready

  Engine.onUpdate = function()
    local p = SceneManager.getPreloadProgress("Level2") -- 0..1
    -- update progress bar with p
    if p >= 1.0 and Input:isKeyDown(Key.SPACE) then
      SceneManager.releaseLoading()
      SceneManager.loadScene("Level2")
      Engine.onUpdate = nil
    end
  end
end

9 — Terrain & water (3D world)

World
lua
local scene = Scene.new()
local cam = Camera.new(scene); cam.position = Vector3.new(0, 18, 22); cam:setTarget(0,0,0); cam:activate()

water.position = Vector3.new(0, 1.5, 0)
water.color = Vector4.new(0.15, 0.45, 0.65, 0.85)
water.waveScale = 0.6

Engine:setScene(scene)
