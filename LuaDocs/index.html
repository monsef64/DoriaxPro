-- Minimal scene: create a sprite and react to input
local scene  = Scene.new()
local player = Sprite.new(scene)
player:setTexture("hero.png")
player.position = Vector3.new(0, 1, 0)
player:setBillboard(true, true, false)

local function update()
  if Input:isKeyDown(Key.SPACE) then
    player:addAction(ScaleAction.new(scene, player)
      :setAction(Ease.easeOutBack(), 1.2, 0.18))
  end
end

Engine.onUpdate = update
Engine:setScene(scene)

Quick start

Full guide →

1. Download or build from source

Grab the latest Windows x64 / Linux x64 release, or build locally:

bash
git clone https://github.com/monsef64/DoriaxPro.git
cd DoriaxPro
cmake -S . -B build -G Ninja -DCMAKE_BUILD_TYPE=Release
cmake --build build --target doriax-editor

2. Script your first scene

In project/lua/main.lua, attach scripts to entities and use the Engine callbacks.

lua
Engine.onViewLoaded = function()
  Log:info("Game started — canvas %dx%d",
    Engine.canvasWidth, Engine.canvasHeight)
end
Engine.onUpdate = function()
  -- game logic runs here every frame
end
💡
Tip: The editor shows live auto-complete for every Lua binding (powered by generate_api_suggestions.py). If it’s missing, re-run cmake --build build --target doriax-editor to regenerate engine_api_suggestions.h.
